Ragged Batching：Attention按请求分块算

Query是本轮新token拼起来的，Key是KV Cache里的历史加上新token，所以Q和K的长度不必相同

def ragged_batching(packed_input, cu_seqlens, batch, kv_cache):
    # 每个请求token数 = [4, 1, 1]
    # → cu_seqlens = [0, 4, 5, 6]

    Q, K, V = qkv_linear(packed_input)

    for i, request in enumerate(batch):
        lo, hi = cu_seqlens[i], cu_seqlens[i + 1]
        kv_cache[request].append(K[lo:hi], V[lo:hi])

    O = empty_like(Q)
    for i, request in enumerate(batch):
        lo, hi = cu_seqlens[i], cu_seqlens[i + 1]
        q = Q[lo:hi]
        state = init_online_softmax()
        for k, v in kv_cache[request].blocks():
            scores = q @ k.T
            state = online_softmax_update(state, scores, v)
        O[lo:hi] = finalize(state)

    return out_linear(O)
